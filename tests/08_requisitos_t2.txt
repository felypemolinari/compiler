program requisitos_t.

:3 Declaracao de variaveis E vetores, com multiplos nomes --

:3 varias variaveis numa declaracao
int:) a, b, c.

:3 varios vetores numa declaracao (tamanho decimal, hexadecimal e binario)
int:) v[10], w[0xFF], z[0b1010].

:3 variaveis e vetores misturados, em todos os tipos
float:) x, notas[5].
string:) nome, nomes[3].
boolean:) ok, flags[4].
char:) letra, buffer[80].

:3 ----------------------------------------------------------------------------
:3 		 Definicao de sub-rotinas (procedimentos e funcoes) com passagem
:3       de parametros -- declaradas aqui porque a gramatica exige sub-rotinas
:3       antes do begin; as chamadas sao testadas no bloco (1,5) la embaixo
:3 ----------------------------------------------------------------------------

:3 funcao com parametro variavel
function int:) dobro(int:) n)
{
  return n * 2.
}

:3 funcao com parametro vetor e parametro variavel
function int:) soma_vetor(int:) vet[], int:) tam)
{
  int:) k, acc.
  acc = 0.
  for (k = 0, k < tam, k = k + 1) do
  {
    acc = acc + vet[k].
  }
  return acc.
}

:3 funcao com varios parametros de tipos diferentes
function float:) media(float:) total, int:) qtd, string:) rotulo)
{
  write(rotulo, total / qtd).
  return total / qtd.
}

:3 procedimento (sem tipo de retorno) com parametros
function mostra(string:) rotulo, int:) valor)
{
  write(rotulo, valor).
  return.
}

:3 procedimento sem parametros
function linha()
{
  write("----------").
}

begin
  :3 --------------------------------------------------------------------------
  :3 Entrada de dados com variaveis e vetores
  :3 --------------------------------------------------------------------------
  
  :3 so variavel
  read(a).
  
  :3 varias variaveis
  read(b, c, nome).
  
  :3 vetor com indice literal, variavel e expressao
  read(v[0], v[a], v[a + 1]).
  
  :3 variaveis e vetores misturados
  read(x, notas[2], letra).

  :3 --------------------------------------------------------------------------
  :3 Saida de dados com variaveis, vetores e literais
  :3 --------------------------------------------------------------------------
  
  :3 so literal (string, char, inteiro, hexa, binario, real, booleano)
  write("literal string").
  write('c', 42, 0x1F, 0b101, 3.14, true, false).

  :3 so variavel
  write(a).
 
  :3 so vetor
  write(v[a]).
  
  :3 tudo misturado, incluindo expressao e chamada
  write("a = ", a, " v[0] = ", v[0], " dobro = ", dobro(a) + 1).

  :3 --------------------------------------------------------------------------
  :3  Atribuicao com variaveis e/ou vetores recebendo expressoes com
  :3       variaveis, vetores, literais e chamadas de sub-rotinas
  :3 --------------------------------------------------------------------------
  
  :3 variavel recebendo literal
  a = 10.

  :3 variavel recebendo variavel
  b = a.
  
	
  :3 variavel recebendo vetor
  c = v[3].
  
  :3 variavel recebendo chamada de sub-rotina
  a = dobro(b).
  
  :3 vetor recebendo literal, variavel, vetor e chamada
  v[0] = 1.
  v[1] = a.
  v[a] = w[b].
  v[a + 1] = soma_vetor(v, 10).
  
  :3 expressao misturando tudo
  a = b + v[2] * dobro(c) - 7.
  x = media(3.0 * x, a, "media: ").
  nome = "texto".
  ok = true.
  letra = 'z'.

  :3 --------------------------------------------------------------------------
  :3    Chamada de sub-rotinas: parametros com variaveis, vetores e
  :3       literais, e chamada como operacao
  :3 --------------------------------------------------------------------------
  
  :3 argumento variavel
  mostra("valor de a: ", a).
  
  :3 argumento vetor (inteiro) e elemento de vetor
  a = soma_vetor(v, v[0]).
  
  :3 argumentos literais
  mostra("literal: ", 99).
  x = media(10.5, 2, "lit").
  
  :3 procedimento sem argumentos
  linha().
  
  :3 chamada como operacao dentro de expressao
  a = dobro(a) + soma_vetor(v, 10) * 2.
  
  :3 chamada como argumento de outra chamada
  a = dobro(dobro(soma_vetor(v, 3))).

  :3 --------------------------------------------------------------------------
  :3 Desvio condicional simples (sem else)
  :3 --------------------------------------------------------------------------
  if (a > 0) then
  {
    write("positivo").
  }

  :3 --------------------------------------------------------------------------
  :3 Desvio condicional composto (com else)
  :3 --------------------------------------------------------------------------
  if (a == b) then
  {
    write("iguais").
  }
  else
  {
    write("diferentes").
  }

  :3 aninhados: if com else dentro de if
  if (ok) then
  {
    if (a < 10) then
    {
      write("menor").
    }
    else
    {
      write("maior ou igual").
    }
  }

  :3 --------------------------------------------------------------------------
  :3 Tres tipos de lacos de repeticao
  :3 --------------------------------------------------------------------------
  
  :3 pre-testado
  a = 0.
  while (a < 10) do
  {
    a = a + 1.
  }

  :3 pre-testado com variavel de controle (inicializacao e pos-operacao)
  for (a = 0, a < 10, a = a + 1) do
  {
    v[a] = a * a.
  }
  
  :3 for com vetor na inicializacao e na pos-operacao
  for (v[0] = 0, v[0] < 5, v[0] = v[0] + 1) do
  {
    write(v[0]).
  }

  :3 pos-testado
  do
  {
    a = a - 1.
  }
  while (a > 0).

  :3 lacos aninhados
  for (a = 0, a < 3, a = a + 1) do
  {
    b = 0.
    while (b < 3) do
    {
      b = b + 1.
    }
  }

  :3 --------------------------------------------------------------------------
  :3 Expressoes aritmeticas, relacionais e logicas sem restricao de tamanho
  :3 e com parenteses, usando variaveis, vetores, sub-rotinas e literais
  :3 --------------------------------------------------------------------------
  
  :3 aritmeticas: + - * / % e unarios
  a = a + b - c * v[0] / 2 % 3.
  a = -a + +b - -c.
  :3 bit a bit: << >> & | ^ ~
  a = (a << 2) | (b >> 1) & ~c ^ 0xFF.
  :3 relacionais: == != < > <= >=
  ok = a == b.
  ok = a != b.
  ok = a < b.
  ok = a > b.
  ok = a <= b.
  ok = a >= b.
  :3 logicas: AND OR NOT
  ok = a > 0 AND b > 0.
  ok = a > 0 OR b > 0.
  ok = NOT ok.
  :3 parenteses aninhados
  a = ((((a + 1) * 2) - (b - 1)) / (c + 1)).
  :3 expressao longa misturando tudo
  ok = NOT (a + v[b * 2] - dobro(c) * (x / 2.5) >= soma_vetor(v, 10) % 7
       AND (b != 0 OR v[0] <= -1)) OR letra == 'a' AND nome != "fim".
end.
