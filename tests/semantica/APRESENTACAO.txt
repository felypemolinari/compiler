program item1.

:3 @@@ BLOCO 2 -----------------------------------------------------------
:3  ITEM 2 (2 pts) -- identificador nao vaza do escopo onde foi declarado.
:3
:3  ESPERADO: 2 erros.
:3    'local'  nao foi declarado no escopo 'global'   (era local de 'f')
:3    'dentro' nao foi declarado no escopo 'global'   (morreu com o bloco)
:3
:3  PARA EXPLICAR: SymbolTable::lookup percorre so os escopos ABERTOS.
:3  O escopo fechado sai da busca mas continua na tabela -- por isso
:3  'local' e 'dentro' ainda aparecem no painel da direita.

program item2.

int:) g.

function int:) f(int:) p)
{
  int:) local.
  local = p * 2.
  return local.
}

begin
  g = f(2).

  if (g > 0) then
  {
    int:) dentro.
    dentro = 1.
    write(dentro).
  }

  :3 erro: 'dentro' so existia dentro do bloco acima
  write(dentro).

  :3 erro: 'local' pertence ao escopo de 'f'

  write(g).

:3 @@@ BLOCO 6 -----------------------------------------------------------
:3  ITEM 7 (2 pts) -- parte A: tipos nas EXPRESSOES.
:3
:3  ESPERADO: 4 erros.
:3    o operador '*' nao se aplica a 'string' e 'string'
:3    o operador '%' exige dois valores int, e recebeu 'float' e 'int'
:3    o operador 'AND' exige dois valores boolean, e recebeu 'int' e 'boolean'
:3    a condicao precisa ser boolean, e nao 'int'
:3
:3  O exemplo do enunciado e exatamente o primeiro: multiplicacao de strings.

program item7a.

int:)     i.
float:)   r.
string:)  s.
boolean:) b.

begin
  i = 1.
  r = 1.5.
  s = "x".
  b = true.

  :3 ok: int e float se misturam, resultado e float
  r = i + r * 2.

  :3 ok: % com dois inteiros
  i = i % 2.

  :3 erro: multiplicacao de strings
  s = s * s.

  :3 erro: % nao aceita float
  i = r % 2.

  :3 erro: AND exige dois boolean
  b = i AND b.

  :3 erro: a condicao de um if precisa ser boolean
  if (i) then
  {
    write("nunca").
  }

  write(i, r, s, b).
end.

program item7b.

:3 @@@ BLOCO 8 -----------------------------------------------------------

program item7c.

:3 @@@ BLOCO 9 -----------------------------------------------------------
:3  "Vetor passado inteiro para uma funcao, funciona?"
:3
:3  ESPERADO: 1 erro, e ele e o interessante.
:3    o vetor 'v' precisa de um indice para ser usado com o operador '+'
:3
:3  EXPLICACAO: ValueType carrega a flag isVector. Um vetor sem indice e
:3  LEGAL como argumento de parametro vetorial (a chamada soma(v, 3) passa)
:3  e ERRO em qualquer outro lugar (v + 1 nao passa).

program extra_vetor.

int:) v[3].
int:) t.

function int:) soma(int:) vet[], int:) n)
{
  return vet[0] + n.
}

begin
  v[0] = 1.

  :3 ok: vetor inteiro como argumento de parametro vetorial
  t = soma(v, 3).

  :3 erro: vetor inteiro no meio de uma conta
  t = v + 1.

  write(t).
end.


:3 @@@ BLOCO 10 ----------------------------------------------------------
:3  "E recursao? A funcao se enxerga?"
:3
:3  ESPERADO: compila, 0 erro, 0 aviso.
:3
:3  EXPLICACAO: a acao 8 insere 'fat' na tabela e SO DEPOIS abre o escopo
:3  e analisa o corpo -- entao 'fat' ja esta visivel quando chama a si mesma.

program extra_recursao.

int:) n.

function int:) fat(int:) k)
{
  if (k <= 1) then
  {
    return 1.
  }
  return k * fat(k - 1).
}

begin
  read(n).
  write(fat(n)).
end.


:3 @@@ BLOCO 11 ----------------------------------------------------------
:3  "Blocos aninhados tem escopo proprio mesmo?"
:3
:3  ESPERADO: compila, 0 erro, 0 aviso.
:3  Na tabela aparecem DUAS linhas chamadas 'a', em escopos diferentes:
:3     a  int  variavel  global/bloco1
:3     a  int  variavel  global/bloco1/bloco2
:3
:3  O write de dentro imprime 0, o de fora imprime n -- escopos distintos.

program extra_blocos.

int:) n.

begin
  read(n).

  while (n > 0) do
  {
    int:) a.
    a = n.

    if (a > 1) then
    {
      int:) a.
      a = 0.
      write(a).
    }

    write(a).
    n = n - 1.
  }
end.
