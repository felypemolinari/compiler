program extra_blocos.

int:) n.

begin
  read(n).

  while (n > 0) do
  {
    int:) a.
    a = n.

    if (a > 1) then
    {
      int:) a.
      a = 0.
      write(a).
    }

    write(a).
    n = n - 1.
  }
end.
