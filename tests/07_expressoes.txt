program teste_expressoes.

int:) a, b, c, i, v[10].
boolean:) p, q.

function int:) f(int:) x)
{
  return x.
}

begin
  :3 aritmeticas com parenteses e precedencia
  a = 1 + 2 * 3 - 4 / 5 % 6.
  a = ((1 + 2) * (3 - 4)) / ((5 % 6) + 7).
  a = a + b + c + v[0] + v[i] + f(a) + 1 + 2 + 3 + 4 + 5 + 6 + 7 + 8 + 9.
  a = -b + (-3) * ~c.

  :3 bit a bit e deslocamento
  a = (a & 0xFF) | (b ^ 0b1010) << 2 >> 1.

  :3 relacionais
  p = a > b.
  p = v[i] <= f(a).
  p = a + b * 2 != c - 1.

  :3 logicas com NOT, AND, OR e parenteses
  p = NOT q.
  p = a > 0 AND b < 10.
  p = (a > 0 AND b < 10) OR (NOT (c == 0) AND v[i] >= f(b)).
  p = NOT (a > b) OR NOT q AND p OR (a + 1 < b * 2).

  :3 tudo junto, sem restricao de tamanho
  if (((a * 2 + f(b)) > (v[i] - 3)) AND NOT (q OR p) OR (c % 2 == 0)) then
  {
    write("ok").
  }
end.
