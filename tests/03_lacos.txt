:3 (2,0) tres tipos de laco de repeticao
program teste_lacos.

int:) i, j, n, v[10].

begin
  :3 1) pre-testado
  i = 0.
  while (i < n) do
  {
    v[i] = 0.
    i = i + 1.
  }

  :3 2) pre-testado com variavel de controle (inicializacao e pos-operacao)
  for (i = 0, i < 10, i = i + 1) do
  {
    v[i] = i * i.
  }

  :3 3) pos-testado
  do
  {
    n = n - 1.
  }
  while (n > 0).

  :3 aninhamento dos tres
  for (i = 0, i < 10, i = i + 1) do
  {
    j = 0.
    while (j < i) do
    {
      do
      {
        j = j + 1.
      }
      while (j < 2).
    }
  }
end.
